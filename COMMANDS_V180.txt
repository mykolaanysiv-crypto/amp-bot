AMP XP / АМПасадори v1.8.0 — короткі команди

# Локально
python -m venv .venv
source .venv/bin/activate
pip install -r requirements.txt
python run_all.py

# QA
python -m compileall -q app scripts tests
pytest -q

# Heroku backup
heroku pg:backups:capture -a YOUR_APP

# Deploy (приклад git workflow)
git add .
git commit -m "AMP XP v1.8.0 Admin UX"
git push heroku main

# Перевірка
heroku ps -a YOUR_APP
heroku logs --tail -a YOUR_APP
