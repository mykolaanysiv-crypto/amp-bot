AMP XP v1.6.5 — БЕЗПЕЧНЕ ОНОВЛЕННЯ HEROKU

# 0. Умови
# Нова папка: ~/Downloads/amp_bot_v165
# Ваша робоча Git-папка: ~/Downloads/amp_bot_v155
# Heroku app: amp-bot-ver-1-5-0

# 1. Перейдіть у робочу папку та переконайтесь, що поточні зміни збережені
cd ~/Downloads/amp_bot_v155
git status

# 2. Обов'язковий backup production PostgreSQL
heroku login
heroku pg:backups:capture -a amp-bot-ver-1-5-0
heroku pg:backups -a amp-bot-ver-1-5-0
heroku pg:backups:download -a amp-bot-ver-1-5-0 -o ~/Downloads/amp_before_v165.dump
ls -lh ~/Downloads/amp_before_v165.dump

# 3. Резервна копія локального .env
cp .env .env.before-v165

# 4. Накладіть v1.6.5 поверх робочої Git-папки. .env/.git/.venv/data не чіпаємо.
rsync -av \
  --exclude='.git' \
  --exclude='.env' \
  --exclude='.venv' \
  --exclude='data' \
  ~/Downloads/amp_bot_v165/ \
  ~/Downloads/amp_bot_v155/

# 5. Оновіть локальні залежності та виконайте preflight
cd ~/Downloads/amp_bot_v155
source .venv/bin/activate
pip install -r requirements.txt
cat VERSION.txt
python -m compileall -q app scripts
git diff --check
# VERSION.txt має показати 1.6.5; compileall/diff --check без виводу = OK.

# 6. Перевірте зміни
git status --short

# 7. Додайте змінені tracked-файли та нові файли v1.6.5
git add -u
git add \
  app/engagement.py \
  app/reports.py \
  app/web/templates/reports.html \
  app/web/templates/goals.html \
  app/web/templates/opportunities.html \
  COMMANDS_V165.txt \
  SERVER_UPDATE_V165.md

git status
# Переконайтесь, що .env та .env.before-v165 НЕ знаходяться у Changes to be committed.

# 8. Commit
git commit -m "AMP XP v1.6.5 reports goals lifecycle opportunities"
git status

# 9. Перевірте Heroku remote
git remote -v
heroku apps:info -a amp-bot-ver-1-5-0

# 10. Deploy. Procfile release автоматично та ідемпотентно оновить схему БД.
git push heroku main

# НЕ запускайте migrate_sqlite_to_postgres.py повторно.

# 11. Перевірка production
heroku ps -a amp-bot-ver-1-5-0
curl https://amp-bot-ver-1-5-0-9632a1434a6d.herokuapp.com/health
# Очікується: {"status":"ok","version":"1.6.5"}

heroku logs --tail -a amp-bot-ver-1-5-0
# Переконайтесь, що немає Traceback / OperationalError / ProgrammingError / IntegrityError.
# Ctrl+C — вийти з tail.

# 12. Після deploy у браузері: Cmd+Shift+R
# Перевірте: Аналітика/PNG, Звіти, Цілі & місії, Можливості, перенесення та скасування тестових сутностей.

# 13. Якщо є критична production-помилка
heroku releases -a amp-bot-ver-1-5-0

# heroku rollback v123 -a amp-bot-ver-1-5-0
