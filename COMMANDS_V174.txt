# AMP XP v1.7.4 — команди production deploy

cd ~/Downloads/amp_bot_v155

git status

# 1. Backup
mkdir -p ~/Downloads/AMP_BACKUPS
heroku pg:backups:capture -a amp-bot-ver-1-5-0
heroku pg:backups -a amp-bot-ver-1-5-0
heroku pg:backups:download -a amp-bot-ver-1-5-0 -o ~/Downloads/AMP_BACKUPS/amp_before_v174.dump
cp .env ~/Downloads/AMP_BACKUPS/.env.before-v174

# 2. Якщо Safari розпакував ZIP у ~/Downloads/amp_bot_v174
rsync -av \
  --exclude='.git' \
  --exclude='.env' \
  --exclude='.venv' \
  --exclude='data' \
  --exclude='.pytest_cache' \
  --exclude='__pycache__' \
  ~/Downloads/amp_bot_v174/ \
  ~/Downloads/amp_bot_v155/

# 3. Локальна перевірка
cd ~/Downloads/amp_bot_v155
source .venv/bin/activate
pip install -r requirements.txt
pip install -r requirements-dev.txt

cat VERSION.txt
cat app/version.py
python -m compileall -q app scripts tests
pytest -q
git diff --check
git status

# VERSION.txt і app/version.py повинні показувати 1.7.4.
# pytest має завершитися успішно.

# 4. Stage
# Спочатку tracked files
git add -u

# Потім нові файли/папки v1.7.4
git add app scripts tests requirements-dev.txt pytest.ini VERSION.txt VERSION_CHECK.txt CHANGELOG.md README.md SERVER_UPDATE.md SERVER_UPDATE_V174.md COMMANDS_V174.txt TEST_REPORT_V174.txt SECURITY.md requirements.txt .env.example

git diff --cached --name-only
git diff --cached --check

# Переконайтесь, що НЕ staged: .env, .env.before-*, *.dump, data/, __pycache__/, *.pyc, .pytest_cache/

# 5. Commit + deploy
git commit -m "AMP XP v1.7.4 reliability and architecture"
git push heroku main
