АМП XP / «АМПасадори» v1.11.1 — ПОЧЕРГОВІ КОМАНДИ HEROKU
================================================================

# 1. Перейти у вже підключений до Heroku Git-репозиторій
cd ~/Downloads/amp_bot_v1103

# 2. Перевірити Git/Heroku remote
git status
git remote -v

# Якщо remote heroku відсутній:
# heroku git:remote -a amp-bot-ver-1-5-0

# 3. Backup production PostgreSQL
heroku pg:backups:capture -a amp-bot-ver-1-5-0
heroku pg:backups -a amp-bot-ver-1-5-0

# 4. Синхронізувати робочу папку з поточним production commit
git fetch heroku main
git reset --hard heroku/main

# 5. Розпакувати v1.11.1 у тимчасову папку
rm -rf /tmp/amp_bot_v1111_release
mkdir -p /tmp/amp_bot_v1111_release
unzip -q ~/Downloads/amp_bot_v1111.zip -d /tmp/amp_bot_v1111_release

# 6. Накласти v1.11.1, не чіпаючи .git/.env/.venv
rsync -av --delete \
  --exclude='.git' \
  --exclude='.env' \
  --exclude='.venv' \
  --exclude='__pycache__' \
  --exclude='.pytest_cache' \
  /tmp/amp_bot_v1111_release/amp_bot_v1111/ \
  ./

# 7. Перевірити версію, секрети та зміни
cat VERSION.txt
git check-ignore .env
git status --short
git diff --stat

# 8. Перевірити MONOBANK_TOKEN БЕЗ показу секрету
if [ -n "$(heroku config:get MONOBANK_TOKEN -a amp-bot-ver-1-5-0)" ]; then
  echo "✅ MONOBANK_TOKEN встановлено"
else
  echo "❌ MONOBANK_TOKEN відсутній"
fi

# 9. Commit
git add -A
git status
git commit -m "Upgrade AMP to v1.11.1 Event Operations & Reporting 2.0"

# 10. Deploy — спочатку БЕЗ --force
git fetch heroku main
git push heroku HEAD:main

# 11. Перевірити release/dyno
heroku releases -a amp-bot-ver-1-5-0
heroku ps -a amp-bot-ver-1-5-0

# 12. Логи
heroku logs --tail -a amp-bot-ver-1-5-0
# Вийти з live logs: Ctrl+C
